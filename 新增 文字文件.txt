<canvas id="game" width="480" height="640"></canvas>

<script>
const canvas = document.getElementById("game");
const ctx = canvas.getContext("2d");

let score, gameOver, scoreSaved;
let player, bullets, enemies;

// 初始化
function initGame() {
  score = 0;
  gameOver = false;
  scoreSaved = false;

  player = { x:220, y:580, w:40, h:20, speed:6 };
  bullets = [];
  enemies = [];

  document.getElementById("score").innerText = "分數：0";
  document.getElementById("rankBtn").style.display = "none";
}

// 排行榜存分數
function saveScore(name, score) {
  const list = JSON.parse(localStorage.getItem("ranking") || "[]");
  list.push({ name: name || "玩家", score });
  list.sort((a,b) => b.score - a.score);
  localStorage.setItem("ranking", JSON.stringify(list.slice(0,10)));
}

// 跳排行榜
function goRank() {
  window.location.href = "leaderboard.html";
}

// 敵人生成
setInterval(() => {
  if (!gameOver) {
    enemies.push({
      x: Math.random() * 440,
      y: -20,
      w: 40,
      h: 20,
      speed: 2 + Math.random() * 2
    });
  }
}, 800);

// 控制
const keys = {};
document.addEventListener("keydown", e => {
  keys[e.code] = true;
  if (e.code === "KeyR") restartGame();
});
document.addEventListener("keyup", e => keys[e.code] = false);

let lastShot = 0;
function shoot() {
  bullets.push({ x: player.x + 18, y: player.y, speed: 8 });
}

// 更新
function update() {
  if (gameOver) return;

  if (keys.ArrowLeft && player.x > 0) player.x -= player.speed;
  if (keys.ArrowRight && player.x < 440) player.x += player.speed;
  if (keys.Space && Date.now() - lastShot > 300) {
    shoot();
    lastShot = Date.now();
  }

  bullets.forEach(b => b.y -= b.speed);
  bullets = bullets.filter(b => b.y > -10);
  enemies.forEach(e => e.y += e.speed);

  enemies.forEach((e, ei) => {
    bullets.forEach((b, bi) => {
      if (b.x < e.x + e.w && b.x + 4 > e.x &&
          b.y < e.y + e.h && b.y + 10 > e.y) {
        enemies.splice(ei, 1);
        bullets.splice(bi, 1);
        score++;
        document.getElementById("score").innerText = "分數：" + score;
      }
    });

    if (player.x < e.x + e.w && player.x + player.w > e.x &&
        player.y < e.y + e.h && player.y + player.h > e.y) {
      gameOver = true;
      document.getElementById("rankBtn").style.display = "inline-block";

      if (!scoreSaved) {
        scoreSaved = true;
        const name = prompt("遊戲結束！請輸入你的名字：", "玩家");
        saveScore(name, score);
      }
    }
  });

  enemies = enemies.filter(e => e.y < 700);
}

// 繪製
function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  ctx.fillStyle = "cyan";
  ctx.fillRect(player.x, player.y, player.w, player.h);

  ctx.fillStyle = "yellow";
  bullets.forEach(b => ctx.fillRect(b.x, b.y, 4, 10));

  ctx.fillStyle = "red";
  enemies.forEach(e => ctx.fillRect(e.x, e.y, e.w, e.h));

  if (gameOver) {
    ctx.fillStyle = "white";
    ctx.font = "40px Arial";
    ctx.fillText("GAME OVER", 100, 320);
    ctx.font = "20px Arial";
    ctx.fillText("R 重新開始 / 點擊排行榜", 100, 360);
  }
}

// 主迴圈
function loop() {
  update();
  draw();
  requestAnimationFrame(loop);
}

// 重新開始
function restartGame() {
  initGame();
}

initGame();
loop();
</script>
